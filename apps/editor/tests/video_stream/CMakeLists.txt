cmake_minimum_required(VERSION 3.24)
project(video_stream_tests LANGUAGES CXX)
if(NOT WIN32)
    message(FATAL_ERROR "These tests exercise Windows overlapped pipe I/O")
endif()
find_program(FFMPEG_EXECUTABLE ffmpeg REQUIRED)
add_executable(video_stream_test video_stream_test.cpp)
target_compile_features(video_stream_test PRIVATE cxx_std_20)
enable_testing()
add_test(NAME video_stream COMMAND video_stream_test "${FFMPEG_EXECUTABLE}" "${CMAKE_CURRENT_BINARY_DIR}/output")
set_tests_properties(video_stream PROPERTIES TIMEOUT 60)
